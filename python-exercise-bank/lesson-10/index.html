制作班级成绩总览

给定三个学生的成绩字典列表。删除不在0—100范围内的成绩，计算每名学生平均分和通过状态，按平均分降序输出排名，最后输出全班最高单科成绩。综合使用列表、字典、循环、判断、函数和排序。 程序使用以下初始数据： students=[{"name":"甲","scores":[80,92,105]},{"name":"乙","scores":[58,67,71]},{"name":"丙","scores":[90,95,88]}] report=summarize(students)

参考答案答案不唯一，仅供参考
# 定义 summarize()，完成题目中的一项独立任务
def summarize(students):
    # 准备“制作班级成绩总览”所需的初始数据
    report=[]
    # 按顺序遍历数据并处理每个元素
    for student in students:
        valid=[s for s in student["scores"] if 0<=s<=100]
        average=sum(valid)/len(valid) if valid else 0
        report.append({"name":student["name"],"average":average,"passed":average>=60,"scores":valid})
    # 把处理结果返回给函数调用处
    return sorted(report,key=lambda x:x["average"],reverse=True)
students=[{"name":"甲","scores":[80,92,105]},{"name":"乙","scores":[58,67,71]},{"name":"丙","scores":[90,95,88]}]
report=summarize(students)
for rank,item in enumerate(report,1):
    # 输出结果，便于检查程序是否正确
    print(rank,item["name"],item["average"],"通过" if item["passed"] else "未通过")
print("最高单科：",max(s for item in report for s in item["scores"]))

参考答案答案不唯一，仅供参考
# 准备“完成库存销售结算”所需的初始数据
inventory={"书":{"price":45,"stock":3},"笔":{"price":6,"stock":10}}
orders=[{"product":"书","quantity":2},{"product":"笔","quantity":12},{"product":"本子","quantity":1}]
success=[]
failed=[]
total=0
# 按顺序遍历数据并处理每个元素
for order in orders:
    name=order["product"]
    quantity=order["quantity"]
    # 根据条件选择需要执行的分支
    if name not in inventory:
        failed.append((name,"商品不存在"))
        continue
    if quantity<=0 or quantity>inventory[name]["stock"]:
        failed.append((name,"数量无效"))
        continue
    subtotal=inventory[name]["price"]*quantity
    inventory[name]["stock"]-=quantity
    total+=subtotal
    success.append((name,quantity,subtotal))
# 输出结果，便于检查程序是否正确
print(success)
print(failed)
print(total)

参考答案答案不唯一，仅供参考
# 准备“实现带重试限制的用户登录”所需的初始数据
users={"alice":"a123","bob":"b456"}
locked={"bob"}
failed={}
# 按顺序遍历数据并处理每个元素
for _ in range(3):
    name=input("用户名：")
    # 根据条件选择需要执行的分支
    if name not in users:
        # 输出结果，便于检查程序是否正确

        print("账户已锁定")
        continue
    if input("密码：")==users[name]:
        print("登录成功")
        break
    failed[name]=failed.get(name,0)+1
    print("密码错误")
    if failed[name]>=3:
        locked.add(name)
else:
    print("登录尝试结束")
print("锁定账户：",locked)

class Student:
    # 定义 __init__()，完成题目中的一项独立任务

        self.name=name
        self.courses=set()

# 定义 Course 类，把相关数据和操作组织在一起
class Course:
    # 定义 __init__()，完成题目中的一项独立任务
    def __init__(self,name,capacity):
        self.name=name
        self.capacity=capacity
        self.students=[]
    # 定义 add_student()，完成题目中的一项独立任务
    def add_student(self,student):
        # 根据条件选择需要执行的分支
        if student in self.students:
            # 把处理结果返回给函数调用处
            return "重复选课"
        if len(self.students)>=self.capacity:
            # 把处理结果返回给函数调用处
            return "课程已满"
        self.students.append(student)
        student.courses.add(self.name)
        # 把处理结果返回给函数调用处
        return "选课成功"
