# 一种参考写法
import csv
def read_scores(filename,class_name):
    result=[]
    try:
        with open(filename,newline="",encoding="utf-8") as f:
            for row in csv.DictReader(f):
                try:result.append({"name":row["name"],"class":class_name,"score":float(row["score"])})
                except ValueError:print("无效成绩：",row)
    except FileNotFoundError:print("找不到：",filename)
    return result

records=read_scores("class_a.csv","A")+read_scores("class_b.csv","B")
records.sort(key=lambda row:row["score"],reverse=True)

    writer=csv.DictWriter(f,fieldnames=["name","class","score"]);writer.writeheader();writer.writerows(records)

import json
def update_settings(filename,updates):
    try:
        with open(filename,encoding="utf-8") as f:data=json.load(f)
    except FileNotFoundError:data={}
    except json.JSONDecodeError:
        print("配置文件损坏");return False
    data.update(updates)
    with open(filename,"w",encoding="utf-8") as f:json.dump(data,f,ensure_ascii=False,indent=2)
    return True

if update_settings("settings.json",{"theme":"dark","font_size":14}):

import csv
groups={}
with open("students.csv",newline="",encoding="utf-8") as f:
    for row in csv.DictReader(f):
        try:row["score"]=float(row["score"])
        except ValueError:
            print("跳过：",row);continue
