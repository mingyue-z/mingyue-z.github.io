# 定义 binary()，完成题目中的一项独立任务
def binary(a,t):
    # 准备“实现迭代二分查找”所需的初始数据
    low,high=0,len(a)-1
    # 在条件满足时重复执行，并确保循环能够结束
    while low<=high:
        mid=(low+high)//2
        # 根据条件选择需要执行的分支

            return mid
        if a[mid]<t:
            low=mid+1
        else:
            high=mid-1
    # 把处理结果返回给函数调用处
    return -1

实现合并两个有序列表

请编写“实现合并两个有序列表”所需的函数。函数形式为 merge(a,b)，使用 0、1、4、2、3 进行调用，并输出返回值或处理后的数据。按题意使用 while 循环并设置明确的结束条件

参考答案答案不唯一，仅供参考
# 定义 merge()，完成题目中的一项独立任务
def merge(a,b):
    # 准备“实现合并两个有序列表”所需的初始数据
    out=[]
    # 在条件满足时重复执行，并确保循环能够结束
    while a and b:
        out.append(a.pop(0) if a[0]<=b[0] else b.pop(0))
    # 把处理结果返回给函数调用处
    return out+a+b
# 输出结果，便于检查程序是否正确
print(merge([1,4],[2,3]))

参考答案答案不唯一，仅供参考
# 定义 ms()，完成题目中的一项独立任务
def ms(a):
    # 根据条件选择需要执行的分支
    if len(a)<=1:
        # 把处理结果返回给函数调用处
        return a
    # 准备“实现递归归并排序”所需的初始数据
    m=len(a)//2
    x=ms(a[:m])
    y=ms(a[m:])
    out=[]
    # 在条件满足时重复执行，并确保循环能够结束
    while x and y:
        out.append(x.pop(0) if x[0]<=y[0] else y.pop(0))
    # 把处理结果返回给函数调用处
    return out+x+y
# 输出结果，便于检查程序是否正确
print(ms([3,1,4,2]))

递归是函数在函数体内调用自身。必须先写“基础情况”让调用停止，再把较大问题缩小为同类的小问题。 请编写“递归生成重组词”所需的函数。函数形式为 perms(s)，使用 1、abc 进行调用，并输出返回值或处理后的数据。按题意使用 for 循环遍历

# 定义 perms()，完成题目中的一项独立任务
def perms(s):
    # 根据条件选择需要执行的分支
    if len(s)<=1:
        # 把处理结果返回给函数调用处
        return [s]
    # 把处理结果返回给函数调用处
    return [s[i]+p for i in range(len(s)) for p in perms(s[:i]+s[i+1:])]
# 输出结果，便于检查程序是否正确
print(perms("abc"))
