给定订单列表 orders = [{"product": "书", "price": 45, "quantity": 2}, {"product": "笔", "price": 6, "quantity": 5}]。编写 calculate_order(orders, discount=1.0)：计算每项小计和折扣后总额，返回“明细列表”和“总额”。以九折调用函数，逐行输出商品小计，最后输出总额。 程序使用以下初始数据： details,total=calculate_order(orders,discount=0.9)

参考答案答案不唯一，仅供参考
# 定义 calculate_order()，完成题目中的一项独立任务
def calculate_order(orders,discount=1.0):
    # 准备“使用函数汇总商品订单”所需的初始数据
    details=[]
    total=0
    # 按顺序遍历数据并处理每个元素
    for order in orders:
        subtotal=order["price"]*order["quantity"]
        details.append({"product":order["product"],"subtotal":subtotal})
        total+=subtotal
    # 把处理结果返回给函数调用处
    return details,total*discount

print("折后总额：",total)

建立模块 survey_utils.py，其中定义 count_choices(answers) 统计选项频次，定义 most_popular(counts) 返回票数最多的选项。在 main.py 中导入这两个函数，统计 ["A", "B", "A", "C", "A", "B"]，输出频次字典和最受欢迎选项。参考答案分别展示两个文件的内容。 开始编程前，先建立以下数据： answers=["A","B","A","C","A","B"] counts=count_choices(answers)

参考答案答案不唯一，仅供参考
# survey_utils.py
# 定义 count_choices()，完成题目中的一项独立任务
def count_choices(answers):
    # 准备“将调查统计函数保存到模块”所需的初始数据
    counts={}
    # 按顺序遍历数据并处理每个元素
    for answer in answers:
        counts[answer]=counts.get(answer,0)+1
    # 把处理结果返回给函数调用处
    return counts
# 定义 most_popular()，完成题目中的一项独立任务
def most_popular(counts):
    # 把处理结果返回给函数调用处
    return max(counts,key=counts.get)
# main.py
# 导入本题需要使用的模块或工具
from survey_utils import count_choices,most_popular
answers=["A","B","A","C","A","B"]
counts=count_choices(answers)
# 输出结果，便于检查程序是否正确
print(counts)
print(most_popular(counts))

# 定义 check_password()，完成题目中的一项独立任务
def check_password(password,banned_words=None):
    # 根据条件选择需要执行的分支
    if banned_words is None:
        # 准备“设计可复用的密码检查函数”所需的初始数据
        banned_words=[]
    reasons=[]
    if len(password)<8:
        reasons.append("长度不足8")
    if not any(ch.isalpha() for ch in password):
        reasons.append("缺少字母")
    if not any(ch.isdigit() for ch in password):
        reasons.append("缺少数字")
    if any(word.lower() in password.lower() for word in banned_words):
        reasons.append("包含禁用词")
    # 把处理结果返回给函数调用处
    return len(reasons)==0,reasons
# 按顺序遍历数据并处理每个元素
for password in ["python","course2026","GoodPass88"]:
    passed,reasons=check_password(password,["course"] )
    # 输出结果，便于检查程序是否正确
    print(password,passed,reasons)

用高阶函数筛选并转换成绩

编写 process_scores(scores, rule, transform) 函数：先用 rule 筛选成绩，再用 transform 转换保留项，返回新列表且不修改原列表。给定 [55, 68, 72, 91]，传入“不低于60”的规则和“加5分但最高100”的转换函数，输出处理结果及原列表。 程序使用以下初始数据： scores=[55,68,72,91] passed=lambda score:score>=60 bonus=lambda score:min(score+5,100)

参考答案答案不唯一，仅供参考
# 定义 process_scores()，完成题目中的一项独立任务
def process_scores(scores,rule,transform):
    # 准备“用高阶函数筛选并转换成绩”所需的初始数据
    result=[]
    # 按顺序遍历数据并处理每个元素
    for score in scores:
        # 根据条件选择需要执行的分支
        if rule(score):
            result.append(transform(score))
    # 把处理结果返回给函数调用处
    return result
scores=[55,68,72,91]
passed=lambda score:score>=60
bonus=lambda score:min(score+5,100)
# 输出结果，便于检查程序是否正确
print(process_scores(scores,passed,bonus))
print(scores)

随机分组并生成组别摘要

给定学生名单 ["甲", "乙", "丙", "丁", "戊", "己"]。编写 make_groups(names, size=2) 函数：使用 random.shuffle() 打乱名单副本，按指定人数切片分组，返回二维列表；不能改变原名单。设置 random.seed(10) 让结果可重复，输出每组编号、成员以及原名单。 开始编程前，先建立以下数据： names=["甲","乙","丙","丁","戊","己"] groups=make_groups(names)

参考答案答案不唯一，仅供参考
# 导入本题需要使用的模块或工具
import random
# 定义 make_groups()，完成题目中的一项独立任务
def make_groups(names,size=2):
    # 准备“随机分组并生成组别摘要”所需的初始数据
    shuffled=names[:]
    random.shuffle(shuffled)
    # 把处理结果返回给函数调用处
    return [shuffled[i:i+size] for i in range(0,len(shuffled),size)]
names=["甲","乙","丙","丁","戊","己"]
random.seed(10)
groups=make_groups(names)
# 按顺序遍历数据并处理每个元素
for number,members in enumerate(groups,1):
    # 输出结果，便于检查程序是否正确
    print("第",number,"组",members)
print("原名单：",names)
