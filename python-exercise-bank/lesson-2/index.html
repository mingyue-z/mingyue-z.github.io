backup[0][1] = 100
print(original)  # [['Python', 100], ['数据库', 85]]
print(backup)

# 浅复制后的两个外层列表仍引用相同的内层列表。
# 若需要完全独立的嵌套对象，可继续学习 copy.deepcopy()。
